% SLIDE 3 (40s): 1D Eulerian vs Lagrangian Compatibility
% ============================================================
\begin{frame}[t]{$(1+1)$D SVE Models and Lagrangian Reconstruction}
\vspace{0.5mm}
\begin{columns}[T,onlytextwidth]
\column{.47\textwidth}
\lead{Eulerian SVE $\mathcal E_1$} \hfill {\scriptsize(Paliathanasis, 2022)}\par
\footnotesize On $h(t,x)>0$, $u(t,x)>0$, real $m\ge 1$:
\[
 \begin{cases}
 h_t+(hu)_x=0,\\
 u_t+uu_x+g(h+B)_x=0,\\
 B_t+\kappa(u^m)_x=0.
 \end{cases}
\]
Mass coordinate $s_x=h$, $s_t=-hu$ defines inverse map $x=\phi(t,s)$ ($\phi_s=1/h>0$, $\phi_t=u>0$).

\column{.49\textwidth}
\lead{Fixed-$x$ bed derivative \& compatible $\mathcal E_L$}\par
\footnotesize Since $b(t,s)=B(t,\phi(t,s))$, chain rule requires
\[
 B_t=\underbrace{b_t}_{\text{fixed }s}-\keyeq{\underbrace{\frac{\phi_t}{\phi_s}b_s}_{\text{advection correction}}}.
\]
With $\mathcal A=\phi_s\phi_{tt}-g\phi_{ss}/\phi_s^2$ and $T_m=\phi_t^{m-1}\phi_{ts}/\phi_s$:
\[
 \keyeq{\boxed{\mathcal E_L[\phi]=D_t\mathcal A-gm\kappa D_sT_m-D_s\!\left(\frac{\phi_t}{\phi_s}\mathcal A\right)=0.}}
\]
\end{columns}
\divider
\footnotesize
\begin{itemize}\setlength{\itemsep}{1.5pt}
 \item \textbf{Discrepancy with printed $\mathcal E_P$ (2026):} $\mathcal E_P$ omits $-D_s((\phi_t/\phi_s)\mathcal A)$. For $\phi=t+s+s^3/6$, $\mathcal E_P[\phi]=0$ identically whereas $\mathcal E_L[\phi](t,0)=g\neq 0$.
 \item \textbf{Exact reconstruction (Prop.\ 2.3):} $\mathcal E_L[\phi]=0\iff\dd\omega_{\mathcal E_L}=0$, recovering single-valued $b(t,s)$ and $(h,u,B)$.
\end{itemize}
\end{frame}

% SLIDE 4 (45s): Parameter-to-Symmetry Mapping
% ============================================================
\begin{frame}[t]{Parameter-to-Symmetry Mapping and Exceptional Exponents}
\vspace{0.5mm}
\begin{columns}[T,onlytextwidth]
\column{.50\textwidth}
\lead{Theorem 2.8: Linear Determining Split}\par
\footnotesize On positive charts for $\mu=(g,\kappa,m)\in(0,\infty)^2\times[1,\infty)$,
\[
 \keyeq{\boxed{\mu\longmapsto\mathfrak g_\mu=\ker\mathscr D_\mu\simeq\ker A_i(m),}}
\]
where free-jet splitting reduces unrestricted smooth generators to finite matrices:
\[
 \begin{aligned}
 A_1(m)=A_L(m)&=(0,0,0,0,\keyeq{m-3}),\\
 A_P(m)&=\begin{pmatrix}0&0&0&0&\keyeq{m-3}&0\\0&0&0&0&0&\keyeq{m-1}\end{pmatrix}.
 \end{aligned}
\]
\column{.46\textwidth}
\lead{Classification for all real $m\ge 1$}\par\smallskip
{\footnotesize
\setlength{\tabcolsep}{3.5pt}
\begin{tabular}{@{}llll@{}}\toprule
Model & Generic ($4$D) & $m=1$ & $m=3$\\\midrule
$\mathcal E_1$ & $\partial_t,\partial_x,\partial_B,t\partial_t+x\partial_x$ & --- & $+X_5$\\
$\mathcal E_L$ & $\partial_t,\partial_\phi,\partial_s,t\partial_t+s\partial_s+\phi\partial_\phi$ & --- & $+Y_5$\\
$\mathcal E_P$ & $Y_1,Y_2,Y_3,Y_4$ & $+t\partial_\phi$ & $+Y_5$\\\bottomrule
\end{tabular}}\par\medskip
\footnotesize
\begin{itemize}\setlength{\itemsep}{1.5pt}
 \item $m=3$: $\widetilde{g\kappa}=g\kappa(L/T)^{3-m}$ becomes scale-invariant.
 \item $m=1$: boost $t\partial_\phi$ is obstructed in $\mathcal E_L$ by $-D_s(\mathcal A/\phi_s)$.
\end{itemize}
\end{columns}
\divider
\smallnote{\textbf{Scope of symbolic splitting:} Linear-exponent powers $u^{am+b}$ collide at finitely many values, whereas transcendental exponents such as $u^{bm-\exp(a-m)}$ break finite monomial splitting in symbolic Lie--ODE reducers.}
\end{frame}

% ============================================================
\begin{frame}[t]{Direct Symbolic First Integrals and Velocity-Flux Check}
\vspace{0.5mm}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\lead{Symbolic order reduction before numerics}\par
\footnotesize Whenever the Euler operator detects an exact derivative $\mathsf E_F(\mathcal E)\equiv 0$, \code{lie\_reduce/ode.py} integrates once symbolically:
\begin{itemize}\setlength{\itemsep}{3pt}
 \item \textbf{Common scaling $Y_4$} ($\phi=tF(s/t)$, $z=s/t$):
 \[
  \keyeq{\frac{\dd}{\dd z}\bigl(L(z)F''\bigr)=0\ \Longrightarrow\ L(z)F''=I_0,}
 \]
 lowering order $3\to 2$ ($L(z)=0\iff$ characteristic ray tangency).
 \item \textbf{Travelling wave} ($\phi=\gamma t+F(s-ct)$):
 \[
  (L_{c,\gamma}(p)p')'=0\ \Longrightarrow\ \int_{p_0}^{p(z)}L_{c,\gamma}(v)\dd v=I(z-z_0).
 \]
\end{itemize}

\column{.48\textwidth}
\lead{General-$Q(u)$ PDE input check (Prop.\ 2.9)}\par
\footnotesize Replace $\kappa u^m$ by any $C^2$ velocity flux $Q(u)$. Time--bed drift $T_\alpha=\partial_t+\alpha\partial_B$ is always admitted and integrates directly to:
\[
 \keyeq{\begin{aligned}
 HU&=C>0,\qquad \frac{U^2}{2g}+H+Z=E,\\
 Q(U(x))&=Q(U_0)-\alpha(x-x_0).
 \end{aligned}}
\]
\begin{itemize}\setlength{\itemsep}{2pt}
 \item Unique $C^2$ profile on the maximal monotone component where $Q'(U)\neq 0$ and $U>0$.
 \item For $Q(u)=\kappa u^m$: $U(x)=(U_0^m-\alpha(x-x_0)/\kappa)^{1/m}$.
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}[t]{$(2+1)$D Planar Sediment-Transport Model $\mathcal E_2$}
\vspace{0.2mm}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\lead{Sloping-bed SVE \& USACE HEC-RAS motivation}\par
\scriptsize
\[
 \begin{aligned}
 h_t+\nabla\cdot(h\mathbf w)&=0,\\
 \mathbf w_t+(\mathbf w\cdot\nabla)\mathbf w+g\nabla(h+B)+f\mathcal J\mathbf w&=0,\\
 B_t+\nabla\cdot\bigl[|\mathbf w|^{m-1}(K\mathbf w-G\nabla B)\bigr]&=0,
 \end{aligned}
\]
where $\mathcal J(u,v)=(-v,u)$ is a quarter-turn (not $\mathbf J_*$). USACE \texttt{HEC-RAS} 2D sediment studies (flushing, dam removal, dredging) motivate $G>0$, $m\in[1.5,3.5]$, $k=K/G\in[2,10]$ without claiming site calibration.\par\smallskip
Under $\langle R,T_c\rangle$, water \& Exner balances integrate to:
\[
 \keyeq{rHV=C_1,\quad rW+\frac{fr^2}{2}=C_2,\quad rq_r+\frac{cr^2}{2}=C_3,}
\]
giving one radial ODE ($\NumPreservedParams$ symbols, no closed-form \texttt{dsolve}):
\[
 \keyeq{\boxed{\Delta_c(r,H)H'=N(r,H),\qquad Z'=P_Z(r,H).}}
\]

\begin{frame}[t]{Hydraulic Criticality and Analytical Solvability Conditions}
\vspace{0.3mm}
\begin{columns}[T,onlytextwidth]
\column{.45\textwidth}
\lead{Physical meaning of $\Delta_c=0$}\par
\footnotesize
\[
 \keyeq{\Delta_c=gH-V^2=0\ \Longleftrightarrow\ \mathrm{Fr}_r=\frac{|V|}{\sqrt{gH}}=1.}
\]
Unit-Froude hydraulic crossing coupled to sediment slope $P_Z$.
\smallskip

\lead{Theorem 3.1 (Two analytic crossings)}\par
\footnotesize At $\Delta_c(r_*,H_*)=N(r_*,H_*)=0$ with $\keyeq{\det\mathbf J_*<0}$:
\begin{itemize}\setlength{\itemsep}{1pt}
 \item At $f=0$, $(1,1)$: $\keyeq{\det\mathbf J_*=-3\cdot 2^{(1-m)/2}\delta<0}$ ($\delta=\gamma_*-\gamma>0$).
 \item Briot--Bouquet index $a=\lambda_o/\lambda-1<-1\notin\mathbb Z_+$ gives \textbf{two $\ln$-free analytic germs} $\ell_\pm$.
 \item \textbf{Two discriminants:} $\operatorname{rank}\mathscr D_\mu$ (available symmetries) vs.\ $\det\mathbf J_*$ (local crossing solvability).
\end{itemize}

\column{.52\textwidth}
\lead{4.1 Solvability Regimes}\par\smallskip
{\scriptsize
\renewcommand{\arraystretch}{1.12}
\setlength{\tabcolsep}{2.5pt}
\begin{tabularx}{\linewidth}{@{}p{.49\linewidth}X@{}}\toprule
Condition & Solution regime\\\midrule
Invertible first integral ($Q'\ne0$) & Closed form (drift; $f>0$ bowl)\\
Monotone quadrature ($L_i\ne0$) & Exact implicit profile\\
$m=3,\ 4g\kappa<1$ & Positive similarity wave ($x^2/t^2$)\\

Briot--Bouquet ($J\cap\mathbb Z_+=\varnothing$) & Convergent singular Taylor\\
Saddle ($\Delta_c=N=0,\det\mathbf J_*<0$) & Two analytic branches $\ell_\pm$\\
$m=3,\ g\kappa=1/4$ & Convergent Laurent ($|\xi|<1/4$)\\

\end{tabularx}}
\end{columns}
\end{frame}

\begin{frame}[t]{Conclusion, Limitations, and Future Work}
\vspace{0.5mm}
\smallnote{\textbf{Takeaway:} Parameters determine available symmetries; selected symmetries determine reduced dimension; analytical conditions determine constructible solutions; classical computation extends them; ML is tested only afterward and has not outperformed classical methods overall.}
\vspace{1.2mm}
\begin{columns}[T,onlytextwidth]
\column{.51\textwidth}
\lead{What Is Proved \& Observed}\par
\footnotesize
\begin{enumerate}\setlength{\itemsep}{2.5pt}
 \item \textbf{Coordinate \& symmetry audit:} Compatible $\mathcal E_L$ restores $-D_s((\phi_t/\phi_s)\mathcal A)$ for bed reconstruction; $A_i(m)a=0$ proves $(1+1)$D jumps occur only at $m=1,3$.
 \item \textbf{Planar sediment reduction \& crossings:} First integrals $(C_1,C_2,C_3)$ reduce $\mathcal E_2$ to $\Delta_c H'=N$, where $\Delta_c=0\iff\mathrm{Fr}_r=1$ yields two analytic branches when $\det\mathbf J_*<0$.
 \item \textbf{Classical superiority:} 1D Chebyshev reaches $\NumChebDiffMed$ depth and $\NumChebZMed$ bed error in $\NumChebMs$\,ms vs.\ $\NumLNNHMed$ / $\NumLNNZMed$ for \texttt{LNN-PINN}.
\end{enumerate}

\column{.45\textwidth}
\lead{What Remains Open}\par
\footnotesize
\begin{itemize}\setlength{\itemsep}{2.5pt}
 \item \textbf{Physical scope:} Open throughflow with $B=Z(r)+ct$; omits bed drag, suspended load, shocks, $\delta\le0$, and calibrated \texttt{HEC-RAS} IBVPs.
 \item \textbf{CPU-only symbolic-ML search:} Propose integrating factors or coordinate charts when non-power sediment laws ($u^{bm-\exp(a-m)}$) break \texttt{Lie}/\texttt{dsolve}.
 \item \textbf{$\ge 4$D models \& parameters:} $(3+1)$D SVE and $\NumPreservedParams+$ preserved symbolic parameters where tensor grids scale as $O(N_r M^p)$.
\end{itemize}
\end{columns}

\end{frame}
